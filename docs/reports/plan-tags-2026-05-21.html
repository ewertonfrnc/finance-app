export function TagBadge({ name, color, size = 'md' }: TagBadgeProps) {
  const isSmall = size === 'sm';
  return (
    <View
      style={{ backgroundColor: color + '33' }}  // 20% opacity via hex
      className={`rounded-full flex-row items-center gap-1
        ${isSmall ? 'px-2 py-0.5' : 'px-2.5 py-1'}`}
    >
      <View
        style={{ backgroundColor: color }}
        className={isSmall ? 'w-1.5 h-1.5 rounded-full'
                           : 'w-2 h-2 rounded-full'}
      />
      <Text
        className={`font-medium text-foreground
          ${isSmall ? 'text-[10px]' : 'text-xs'}`}
      >{name}</Text>
    </View>
  );
}

typescript src/features/tags/hooks/useInvalidateTagData.ts — novo hook 
export function useInvalidateTagData() {
  const queryClient = useQueryClient();
  const userId = useAuthStore((s) => s.userId);

  return () =>
    Promise.all([
      queryClient.invalidateQueries({
        queryKey: queryKeys.tagsAll(userId),
      }),
      queryClient.invalidateQueries({
        queryKey: queryKeys.transactionsAll(userId),
        predicate: (q) => !q.queryKey.includes("detail"),
      }),
      // ⚠ NÃO invalida balance — tags não afetam saldo
    ]);
}

typescript Batch fetch de tags — evitar N+1 
// ✓ Uma query para M transações — batch por IDs
export async function getTagsByTransactionIds(
  txIds: string[]
): Promise<Record<string, ApiTag[]>> {
  if (!txIds.length) return {};

  // Backend: SELECT ... WHERE tt.transaction_id = ANY($1::uuid[])
  const res = await apiClient.post<ApiResponse<ApiTagsByTx[]>>(
    "/v1/tags/batch", { transaction_ids: txIds }
  );
  // Agrupar por transaction_id no cliente
  return res.data.data.reduce((acc, row) => {
    (acc[row.transaction_id] ??= []).push(row);
    return acc;
  }, {} as Record<string, ApiTag[]>);
}

// ✗ Padrão N+1 — NUNCA fazer isso
for (const tx of transactions) {
  tx.tags = await getTagsForTransaction(tx.id); // 1 query por transação
}
typescript setTransactionTags — semântica replace-all 
// O backend faz DELETE + INSERT em uma SQL tx.
// Enviar [] remove todas as tags. Enviar lista completa substitui tudo.
export async function setTransactionTags(
  transactionId: string,
  tagIds: string[]        // lista COMPLETA desejada, não delta
): Promise<void> {
  await apiClient.put(
    `/v1/transactions/${transactionId}/tags`,
    { tags: tagIds }
  );
}
